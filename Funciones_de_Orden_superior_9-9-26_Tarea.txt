def convertir_temperatura(valor, conversion):
    return conversion(valor)

def c_a_f(valor):
    return (valor * 9 / 5) + 32

def f_a_c(valor):
    return (valor - 32) * 5 / 9

def c_a_k(valor):
    return valor + 273.15

temperatura_c = 30
print("Temperatura en Celsius:", temperatura_c)
print("Celsius a Fahrenheit:", convertir_temperatura(temperatura_c, c_a_f))
print("Celsius a Kelvin:", convertir_temperatura(temperatura_c, c_a_k))

temperatura_f = 86
print("Fahrenheit a Celsius:", convertir_temperatura(temperatura_f, f_a_c))
